/********************************************************************
 * GIL - MATRICE DE COUVERTURE DES EXIGENCES
 * Postman / Newman
 *
 * Objectif :
 * Jira EF / Critere -> Controle -> Resultat -> Preuve / Commentaire
 *
 * Statuts :
 * PASS / FAIL          = controle reellement execute
 * NOT AUTOMATABLE      = controle impossible avec Postman seul
 * NOT EXECUTED         = necessite un scenario Postman dedie
 ********************************************************************/


/********************************************************************
 * 0. PARAMETRAGE
 ********************************************************************/

const CONFIG = {

    // Codes consideres comme succes pour le scenario nominal
    successCodes: [200, 201, 202, 204],

    // A adapter au SLA reel
    slaMs: 2000,

    /*
     * Scenario de la requete courante.
     *
     * Valeurs possibles :
     * NOMINAL
     * JWT_MISSING
     * JWT_INVALID
     * PAYLOAD_EMPTY
     * BUSINESS_ERROR
     * TIMEOUT
     * IDEMPOTENCY_REPLAY
     */
    scenario:
        pm.variables.get("testScenario") ||
        pm.environment.get("testScenario") ||
        "NOMINAL"
};


console.log("======================================================");
console.log("GIL - CONTROLE DES EXIGENCES");
console.log("Scenario : " + CONFIG.scenario);
console.log("======================================================");


/********************************************************************
 * 1. FONCTIONS GENERIQUES
 ********************************************************************/

function getRequestHeader(name) {
    try {
        return pm.request.headers.get(name);
    } catch (e) {
        return null;
    }
}


function getResponseHeader(name) {
    try {
        return pm.response.headers.get(name);
    } catch (e) {
        return null;
    }
}


function addEvidence(id, message) {
    console.log("[PREUVE] " + id + " | " + message);
}


function addNote(id, message) {
    console.warn("[NOTE] " + id + " | " + message);
}


/*
 * Permet d'afficher dans Newman une exigence
 * non automatisable sans la declarer artificiellement PASS.
 */
function notAutomatable(id, title, comment) {

    const testName =
        id + " | NOT AUTOMATABLE | " + title;

    if (
        pm.test &&
        typeof pm.test.skip === "function"
    ) {

        pm.test.skip(testName, function () {});

    }

    addNote(
        id,
        "NOT AUTOMATABLE - " + comment
    );
}


/*
 * Exigence qui pourrait etre automatisee,
 * mais necessite une requete/scenario specifique.
 */
function notExecuted(id, title, comment) {

    const testName =
        id + " | NOT EXECUTED | " + title;

    if (
        pm.test &&
        typeof pm.test.skip === "function"
    ) {

        pm.test.skip(testName, function () {});

    }

    addNote(
        id,
        "NOT EXECUTED - " + comment
    );
}


/*
 * Test standard avec preuve.
 */
function requirementTest(id, title, testFunction) {

    pm.test(
        id + " | " + title,
        function () {
            testFunction();
        }
    );
}

 * EF-01
 * RECEPTION DE LA REQUETE API

console.log("");
console.log("======================================================");
console.log("EF-01 - RECEPTION DE LA REQUETE API");
console.log("======================================================");


/*
 * EF-01.01
 * Endpoint expose conformement au contrat.
 */

requirementTest(
    "EF-01.01",
    "Endpoint API accessible",
    function () {

        pm.expect(
            pm.response.code,
            "Aucune reponse HTTP exploitable"
        ).to.be.a("number");

        addEvidence(
            "EF-01.01",
            pm.request.method +
            " " +
            pm.request.url.toString() +
            " -> HTTP " +
            pm.response.code
        );
    }
);


/*
 * EF-01.02
 * Endpoint mappe dans APIGEE.
 *
 * Postman voit le resultat E2E,
 * mais pas la configuration interne APIGEE.
 */

notAutomatable(
    "EF-01.02",
    "Endpoint mappe dans APIGEE",
    "La configuration et le mapping APIGEE ne sont pas visibles " +
    "depuis Postman. Preuve attendue : configuration APIGEE ou logs APIGEE."
);


/*
 * EF-01.03
 * Rejet des requetes sans headers obligatoires.
 */

if (
    CONFIG.scenario === "JWT_MISSING" ||
    CONFIG.scenario === "JWT_INVALID"
) {

    requirementTest(
        "EF-01.03",
        "Rejet d'une requete non authentifiee",
        function () {

            pm.expect(
                pm.response.code,
                "HTTP 401 attendu"
            ).to.eql(401);

            addEvidence(
                "EF-01.03",
                "HTTP " + pm.response.code
            );
        }
    );

} else {

    notExecuted(
        "EF-01.03",
        "Rejet des requetes sans headers obligatoires",
        "Necessite une requete negative dediee " +
        "avec header obligatoire absent."
    );
}


/*
 * EF-01.04
 * Correlation ID present.
 */

const correlationId =
    getRequestHeader("X-Correlation-Id") ||
    getRequestHeader("x-correlation-id") ||
    getRequestHeader("correlationId");


requirementTest(
    "EF-01.04",
    "Identifiant de correlation present",
    function () {

        pm.expect(
            correlationId,
            "Correlation ID absent"
        ).to.exist;

        pm.expect(
            String(correlationId).trim(),
            "Correlation ID vide"
        ).to.not.eql("");

        addEvidence(
            "EF-01.04",
            "CorrelationId=" + correlationId
        );
    }
);


/*
 * EF-01.05
 * Propagation correlation ID dans tous les logs.
 */

notAutomatable(
    "EF-01.05",
    "Propagation du CorrelationId dans tous les logs",
    "Postman ne peut pas consulter tous les logs internes. " +
    "Utiliser le CorrelationId [" +
    (correlationId || "NON DISPONIBLE") +
    "] dans Argo / Kibana / Elastic."
);


/********************************************************************
 *
 * EF-02
 * TRANSMISSION DE LA REQUETE API
 *
 ********************************************************************/

console.log("");
console.log("======================================================");
console.log("EF-02 - TRANSMISSION DE LA REQUETE API");
console.log("======================================================");


/*
 * EF-02.01
 * Transmission GIL -> APIGEE / backend.
 */

if (CONFIG.scenario === "NOMINAL") {

    requirementTest(
        "EF-02.01",
        "Traitement E2E observable",
        function () {

            pm.expect(
                pm.response.code,
                "Code HTTP nominal inattendu"
            ).to.be.oneOf(CONFIG.successCodes);

            addEvidence(
                "EF-02.01",
                "Reponse finale recue : HTTP " +
                pm.response.code
            );
        }
    );

} else {

    notExecuted(
        "EF-02.01",
        "Traitement E2E nominal",
        "Le scenario courant est " +
        CONFIG.scenario +
        ", pas NOMINAL."
    );
}


/*
 * EF-02.02
 * Routage exact GIL -> APIGEE -> Entite.
 */

notAutomatable(
    "EF-02.02",
    "Routage interne GIL vers le destinataire",
    "Postman ne voit pas les hops internes. " +
    "Preuve attendue dans les logs GIL/APIGEE/backend."
);


/*
 * EF-02.03
 * Propagation des headers de tracabilite.
 */

requirementTest(
    "EF-02.03",
    "Headers de tracabilite presents",
    function () {

        pm.expect(
            correlationId,
            "X-Correlation-Id absent"
        ).to.exist;

        addEvidence(
            "EF-02.03",
            "X-Correlation-Id=" + correlationId
        );
    }
);


/*
 * Idempotency-Key
 */

const idempotencyKey =
    getRequestHeader("Idempotency-Key") ||
    getRequestHeader("idempotency-key");


if (idempotencyKey) {

    requirementTest(
        "EF-02.04",
        "Idempotency-Key present",
        function () {

            pm.expect(
                String(idempotencyKey).trim()
            ).to.not.eql("");

            addEvidence(
                "EF-02.04",
                "Idempotency-Key=" +
                idempotencyKey
            );
        }
    );

} else {

    notExecuted(
        "EF-02.04",
        "Idempotency-Key",
        "Header non present sur cette requete. " +
        "Verifier si l'operation courante est concernee par l'idempotence."
    );
}


/*
 * EF-02.05
 * Transformation du payload.
 */

notAutomatable(
    "EF-02.05",
    "Transformation du payload",
    "La transformation interne n'est pas observable directement " +
    "dans Postman sauf si les donnees transformees sont restituees " +
    "par une API GET. Sinon preuve via logs ou comparaison cote cible."
);


/********************************************************************
 *
 * EF-03
 * REPONSE A LA REQUETE API
 *
 ********************************************************************/

console.log("");
console.log("======================================================");
console.log("EF-03 - REPONSE A LA REQUETE API");
console.log("======================================================");


/*
 * EF-03.01
 * Retour synchrone avec code HTTP correspondant.
 */

if (CONFIG.scenario === "NOMINAL") {

    requirementTest(
        "EF-03.01",
        "Code HTTP nominal conforme",
        function () {

            pm.expect(
                pm.response.code
            ).to.be.oneOf(CONFIG.successCodes);

            addEvidence(
                "EF-03.01",
                "HTTP " + pm.response.code
            );
        }
    );

} else {

    addNote(
        "EF-03.01",
        "Controle nominal non execute car scenario=" +
        CONFIG.scenario
    );
}


/*
 * EF-03.02
 * Temps de reponse.
 */

requirementTest(
    "EF-03.02",
    "Temps de reponse conforme au seuil",
    function () {

        pm.expect(
            pm.response.responseTime,
            "Temps=" +
            pm.response.responseTime +
            " ms"
        ).to.be.below(CONFIG.slaMs);

        addEvidence(
            "EF-03.02",
            pm.response.responseTime +
            " ms / seuil=" +
            CONFIG.slaMs +
            " ms"
        );
    }
);


/*
 * EF-03.03
 * Content-Type.
 */

const contentType =
    getResponseHeader("Content-Type") || "";


if (pm.response.code !== 204) {

    requirementTest(
        "EF-03.03",
        "Content-Type de la reponse present",
        function () {

            pm.expect(
                contentType,
                "Content-Type absent"
            ).to.not.eql("");

            addEvidence(
                "EF-03.03",
                "Content-Type=" + contentType
            );
        }
    );
}


/*
 * EF-03.04
 * JSON valide si la reponse est JSON.
 */

if (
    contentType
        .toLowerCase()
        .includes("application/json")
) {

    requirementTest(
        "EF-03.04",
        "Body JSON valide",
        function () {

            pm.expect(function () {
                pm.response.json();
            }).to.not.throw();

            addEvidence(
                "EF-03.04",
                "JSON parse avec succes"
            );
        }
    );

} else {

    notExecuted(
        "EF-03.04",
        "Validation du JSON",
        "La reponse courante n'est pas application/json."
    );
}


/*
 * EF-03.05
 * Erreurs 4xx / 5xx.
 */

if (
    CONFIG.scenario === "BUSINESS_ERROR" ||
    CONFIG.scenario === "TIMEOUT"
) {

    requirementTest(
        "EF-03.05",
        "Propagation d'une erreur 4xx/5xx",
        function () {

            pm.expect(
                pm.response.code
            ).to.be.within(400, 599);

            addEvidence(
                "EF-03.05",
                "HTTP " + pm.response.code
            );
        }
    );

} else {

    notExecuted(
        "EF-03.05",
        "Gestion des erreurs destinataire",
        "Necessite un scenario negatif dedie provoquant " +
        "une erreur 4xx/5xx cote destinataire."
    );
}


/*
 * EF-03.06
 * Pas de divulgation d'informations internes.
 */

if (pm.response.code >= 400) {

    requirementTest(
        "EF-03.06",
        "Absence de fuite technique evidente",
        function () {

            const body =
                pm.response.text().toLowerCase();

            const forbiddenPatterns = [
                "stacktrace",
                "stack trace",
                "java.lang.",
                "nullpointerexception",
                "sql exception",
                "private key",
                "database password",
                "password=",
                "secret="
            ];

            forbiddenPatterns.forEach(
                function (pattern) {

                    pm.expect(
                        body,
                        "Contenu technique detecte : " +
                        pattern
                    ).to.not.include(pattern);

                }
            );

            addEvidence(
                "EF-03.06",
                "Aucun pattern technique sensible connu detecte"
            );
        }
    );

} else {

    notExecuted(
        "EF-03.06",
        "Non-divulgation d'informations internes",
        "Controle pertinent principalement sur les scenarios d'erreur."
    );
}

            pm.expect(

                "HTTP 401"

} else if (CONFIG.scenario === "JWT_INVALID") {

    requirementTest(
        "CA-01",
        "JWT invalide -> HTTP 401",
        function () {

            pm.expect(

            ).to.eql(401);

            addEvidence(
                "CA-01",
                "HTTP 401"
            );
        }
    );

} else {

    notExecuted(
        "CA-01",
        "Controle JWT invalide/absent",
        "Creer les scenarios JWT_MISSING et JWT_INVALID."
    );

/*
 * CA-02 Payload manquant ou vide -> 400.
 */

if (CONFIG.scenario === "PAYLOAD_EMPTY") {

    requirementTest(
        "CA-02",
        "Payload vide -> HTTP 400",
        function () {

            pm.expect(
                pm.response.code
            ).to.eql(400);

            addEvidence(
                "CA-02",
                "HTTP 400"
            );
        }
    );

} else {

    notExecuted(
        "CA-02",
        "Validation payload vide",
        "Necessite une requete PAYLOAD_EMPTY sans body."
    );
}


/*
 * CA-03 Transformation du payload.
 */

    "CA-03",
    "Transformation du payload conforme",
    "Necessite l'observation du payload transforme cote destinataire " +

 * CA-04 Routage vers entite interne.

    "CA-04",

);

 * CA-05 Gestion de la reponse.
 */

requirementTest(
    "CA-05",
    "Une reponse HTTP est restituee au consommateur",
    function () {

        pm.expect(
            pm.response.code
        ).to.be.a("number");

        addEvidence(
            "CA-05",
            "HTTP " +
            pm.response.code +
            " / " +
            pm.response.responseTime +
            " ms"
        );
    }
);


/*
 * CA-06 Erreurs metier.
 */

if (CONFIG.scenario === "BUSINESS_ERROR") {

    requirementTest(
        "CA-06",
        "Erreur metier correctement restituee",
        function () {

            pm.expect(
                pm.response.code
            ).to.be.within(400, 599);

            addEvidence(
                "CA-06",
                "HTTP " + pm.response.code
            );
        }
    );

} else {

    notExecuted(
        "CA-06",
        "Gestion des erreurs metier",
        "Necessite un scenario BUSINESS_ERROR."
    );
}


/*
 * CA-07 Idempotence.
 */

if (CONFIG.scenario === "IDEMPOTENCY_REPLAY") {

    requirementTest(
        "CA-07",
        "Rejeu avec le meme Idempotency-Key",
        function () {

            pm.expect(
                idempotencyKey,
                "Idempotency-Key absent"
            ).to.exist;

            pm.expect(
                pm.response.code
            ).to.be.oneOf(
                [200, 201, 202, 204, 409]
            );

            addEvidence(
                "CA-07",
                "Rejeu execute avec key=" +
                idempotencyKey +
                " / HTTP " +
                pm.response.code
            );
        }
    );

} else {

    notExecuted(
        "CA-07",
        "Idempotence",
        "Executer deux fois la meme operation avec le meme " +
        "Idempotency-Key et comparer le resultat fonctionnel."
    );
}
