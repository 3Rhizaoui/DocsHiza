grep -n -E \

echo
echo "===== JSON GENERES ====="
ls -lh \
portal/commun/data/gil_project.json \

2>/dev/null

echo
echo "===== KPI PROJECT / HOME ====="

python - <<'PY'
import json
from pathlib import Path

base = Path("portal/commun/data")

for name in [
    "gil_project.json",
    "gil_home.json",
]:
    p = base / name

    print()
    print("====", name, "====")

    if not p.exists():
        print("ABSENT")
        continue

    d = json.loads(
        p.read_text(
            encoding="utf-8"
        )
    )

    print(
        "generatedAt =",
        d.get("generatedAt")
    )

    print(
        "arrimage =",
        d.get("arrimage")
    )

    print(
        "santeFluxArrimage =",
        d.get("santeFluxArrimage")
    )
PY
