Dans le même cmd, exécute ce bloc :
echo ===== EDGE 64 BITS =====
if exist "%ProgramFiles%\Microsoft\Edge\Application\msedge.exe" echo %ProgramFiles%\Microsoft\Edge\Application\msedge.exe

echo ===== EDGE 32 BITS =====
if exist "%ProgramFiles(x86)%\Microsoft\Edge\Application\msedge.exe" echo %ProgramFiles(x86)%\Microsoft\Edge\Application\msedge.exe

echo ===== EDGE UTILISATEUR =====
if exist "%LOCALAPPDATA%\Microsoft\Edge\Application\msedge.exe" echo %LOCALAPPDATA%\Microsoft\Edge\Application\msedge.exe

Puis :
echo ===== REGISTRE EDGE =====
reg query "HKLM\SOFTWARE\Microsoft\Windows\CurrentVersion\App Paths\msedge.exe" /ve

reg query "HKCU\SOFTWARE\Microsoft\Windows\CurrentVersion\App Paths\msedge.exe" /ve

Si les commandes registre ne donnent rien, utilise cette recherche ciblée :
dir "C:\Program Files\Microsoft\Edge\Application\msedge.exe" /s /b 2>nul

dir "C:\Program Files (x86)\Microsoft\Edge\Application\msedge.exe" /s /b 2>nul

dir "%LOCALAPPDATA%\Microsoft\Edge\Application\msedge.exe" /s /b 2>nul
