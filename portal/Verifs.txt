pm.test("Statut HTTP acceptable", function () {
    pm.expect(pm.response.code).to.be.oneOf([200, 201, 202, 204]);
});

const body = pm.response.text();

if (body && body.trim() !== "") {
    try {
        const jsonData = pm.response.json();
        console.log("Réponse JSON reçue", jsonData);
    } catch (e) {
        console.log("Réponse reçue mais non JSON");
    }
} else {
    console.log("Réponse vide : pas de parsing JSON");
}
