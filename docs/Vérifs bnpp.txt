python - <<'PY'
import json
from pathlib import Path

p = Path(
    "portal/commun/data/standalone/jira/"
    "capabilities_brut.json"
)

d = json.loads(
    p.read_text(
        encoding="utf-8",
        errors="replace"
    )
)

rows = d.get("capabilities", [])

print("=" * 100)
print("STRUCTURE COMPLETE D'UNE FEATURE REELLE")
print("=" * 100)

if not rows:
    raise SystemExit("Aucune capability")

for i, row in enumerate(rows[:2], 1):
    print()
    print("#" * 100)
    print("FEATURE", i)
    print("#" * 100)

    print(
        json.dumps(
            row,
            ensure_ascii=False,
            indent=2
        )[:20000]
    )

print()
print("=" * 100)
print("FIELDS DECLARES PAR L'EXTRACTEUR")
print("=" * 100)

print(
    json.dumps(
        d.get("fields", {}),
        ensure_ascii=False,
        indent=2
    )[:20000]
)

print()
print("=" * 100)
print("BUSINESS RULES")
print("=" * 100)

print(
    json.dumps(
        d.get("businessRules", {}),
        ensure_ascii=False,
        indent=2
    )[:10000]
)
PY
