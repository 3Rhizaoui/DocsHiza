python - <<'PY'
import json
from pathlib import Path

p = Path(
    "portal/commun/data/standalone/jira/"
    "capabilities.json"
)

data = json.loads(
    p.read_text(
        encoding="utf-8"
    )
)

rows = data.get(
    "capabilities"
) or []

print(
    "CAPABILITIES NORMALISEES =",
    len(rows)
)

for row in rows:

    if row.get("jiraKey") != "AERL_GIL-206":
        continue

    print()
    print(
        "KEY         =",
        row.get("jiraKey")
    )
    print(
        "CAPABILITY  =",
        row.get("capability")
    )
    print(
        "VERSION     =",
        repr(
            row.get("version")
        )
    )
    print(
        "ENV         =",
        repr(
            row.get("environnement")
        )
    )

    readiness = (
        row.get("jiraReadiness")
        or {}
    )

    print(
        "TOTAL       =",
        readiness.get("total")
    )
    print(
        "READY       =",
        readiness.get(
            "readyForTest"
        )
    )
    print(
        "NON READY   =",
        readiness.get(
            "nonReady"
        )
    )
    print(
        "PRET        =",
        readiness.get("pret")
    )

    print()
    print("=== TACHES ===")

    for task in (
        row.get("taches")
        or []
    ):
        print(
            task.get("jiraKey"),
            "|",
            task.get("statut"),
            "| ready =",
            task.get(
                "readyForTest"
            ),
            "|",
            task.get("titre")
        )

    break
PY
