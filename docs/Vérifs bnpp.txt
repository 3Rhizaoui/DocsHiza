Puis immédiatement :

python - <<'PY'
import json
from pathlib import Path

p = Path(
    "portal/commun/data/standalone/"
    "payload_standalone.json"
)

data = json.loads(
    p.read_text(
        encoding="utf-8",
        errors="replace"
    )
)

rows = data.get("capabilities") or []

print("TOTAL PAYLOAD =", len(rows))

for row in rows:

    if row.get("jiraKey") != "AERL_GIL-206":
        continue

    print()
    print("==============================")
    print("AERL_GIL-206 DANS LE PAYLOAD")
    print("==============================")

    print(
        "sourceType =",
        row.get("sourceType")
    )

    print(
        "capability =",
        repr(row.get("capability"))
    )

    print(
        "version =",
        repr(row.get("version"))
    )

    print(
        "environnement =",
        repr(row.get("environnement"))
    )

    jira = row.get("jira") or {}

    print()
    print("JIRA =")
    print(
        json.dumps(
            jira,
            ensure_ascii=False,
            indent=2
        )
    )

    print()
    print("READINESS DIRECT =")
    print(
        json.dumps(
            row.get("readiness"),
            ensure_ascii=False,
            indent=2
        )
    )

    break
else:
    print("AERL_GIL-206 INTROUVABLE")
PY
