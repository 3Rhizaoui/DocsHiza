python - <<'PY'
import json
from pathlib import Path

p = Path(
    "portal/commun/data/standalone/"
    "payload_standalone.json"
)

data = json.loads(
    p.read_text(
        encoding="utf-8",
        errors="replace"
    )
)

rows = (
    data.get("rows")
    or data.get("qualifications")
    or []
)

print("TOTAL =", len(rows))

for i, row in enumerate(rows[:8]):

    print()
    print("=" * 70)
    print("ROW", i)
    print("=" * 70)

    print("sourceType =", row.get("sourceType"))
    print("capability =", row.get("capability"))
    print("titre =", row.get("titre"))
    print("jiraKey =", row.get("jiraKey"))

    jira = row.get("jira") or {}
    octane = row.get("octane") or {}

    print()
    print("JIRA:")
    print("  summary =", jira.get("summary"))
    print("  titre =", jira.get("titre"))
    print("  capability =", jira.get("capability"))
    print("  version =", jira.get("version"))
    print("  environnement =", jira.get("environnement"))

    print()
    print("OCTANE:")
    print("  capability =", octane.get("capability"))
    print(
        "  feature nom =",
        (octane.get("octaneFeature") or {}).get("nom")
    )
    print(
        "  test suite =",
        (octane.get("testSuite") or {}).get("nom")
    )
