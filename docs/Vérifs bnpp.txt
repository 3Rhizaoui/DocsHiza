Avant de patcher, vérifions précisément où les 12 autres sont perdues. Sur le PC BNP, lance :

python - <<'PY'
import json
from pathlib import Path

files = [
    Path("portal/commun/data/standalone/octane/qualifications_brut.json"),
    Path("portal/commun/data/standalone/octane/qualifications.json"),
]

for p in files:
    print()
    print("=" * 70)
    print(p)
    print("=" * 70)

    if not p.exists():
        print("ABSENT")
        continue

    d = json.loads(
        p.read_text(
            encoding="utf-8",
            errors="replace"
        )
    )

    print("CLES =", list(d.keys()))

    rows = (
        d.get("qualifications")
        or d.get("features")
        or d.get("items")
        or []
    )

    print("COUNT =", len(rows))

    for r in rows:
        feature = r.get("octaneFeature") or {}

        print(
            "- capability =",
            r.get("capability"),
            "| feature =",
            feature.get("nom"),
            "| featureId =",
            feature.get("id"),
            "| executions =",
            len(r.get("executions") or []),
        )
PY
