Et enfin, pour vérifier les vraies quantités d'entrée :

python - <<'PY'
import json
from pathlib import Path

base = Path("portal/commun/data/standalone")

for rel, key in [
    ("jira/capabilities.json", "capabilities"),
    ("octane/qualifications.json", "qualifications"),
]:
    p = base / rel
    data = json.loads(
        p.read_text(
            encoding="utf-8",
            errors="replace"
        )
    )
    rows = data.get(key) or []

    print()
    print(rel)
    print("COUNT =", len(rows))

    for r in rows:
        print(
            "-",
            r.get("capability"),
            "|",
            r.get("version")
            or r.get("release"),
            "|",
            r.get("environnement")
        )
PY
