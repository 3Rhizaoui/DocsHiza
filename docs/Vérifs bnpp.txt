python - <<'PY'
import json
from pathlib import Path

paths = [
    Path("portal/commun/data/standalone/jira/capabilities.json"),
    Path("portal/commun/data/standalone/jira/capabilities_brut.json"),
    Path("portal/commun/data/standalone/payload_standalone.json"),
]

for p in paths:
    print()
    print("=" * 100)
    print(p)
    print("=" * 100)

    if not p.exists():
        print("ABSENT")
        continue

    d = json.loads(
        p.read_text(
            encoding="utf-8",
            errors="replace"
        )
    )

    if isinstance(d, dict):
        print("CLES =", list(d.keys()))

        for key in (
            "capabilities",
            "features",
            "issues",
            "items",
        ):
            rows = d.get(key)

            if isinstance(rows, list):
                print(key, "COUNT =", len(rows))

                for r in rows[:20]:
                    if isinstance(r, dict):
                        epic = r.get("epic") or {}

                        print(
                            "- jiraKey =",
                            r.get("jiraKey")
                            or epic.get("key"),
                            "| capability =",
                            r.get("capability"),
                            "| summary =",
                            (epic.get("fields") or {}).get("summary")
                        )
PY
