Puis affiche seulement les éléments utiles :

python - <<'PY'
from pathlib import Path
import json

p = Path(
    "portal/commun/data/standalone/octane/"
    "octane_hierarchy_diagnostic.json"
)

data = json.loads(
    p.read_text(
        encoding="utf-8",
        errors="replace",
    )
)

print("EPIC")
print(
    data.get("epic", {}).get("id"),
    "|",
    data.get("epic", {}).get("name"),
)

print()
print(
    "FEATURES =",
    len(data.get("features") or [])
)

for row in data.get("features") or []:
    print(
        row.get("id"),
        "|",
        row.get("name"),
    )

print()
print("FEATURE SELECTIONNEE")
row = data.get("selectedFeature") or {}

print(
    row.get("id"),
    "|",
    row.get("name"),
)
PY
