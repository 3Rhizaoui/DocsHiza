import json
from pathlib import Path

p = Path(
    "portal/commun/data/standalone/"
    "payload_standalone.json"
)

data = json.loads(
    p.read_text(
        encoding="utf-8",
        errors="replace"
    )
)

print("TYPE RACINE =", type(data).__name__)
print("CLES RACINE =", list(data.keys()))

for key, value in data.items():
    if isinstance(value, list):
        print(
            key,
            "=> LISTE",
            len(value)
        )
    elif isinstance(value, dict):
        print(
            key,
            "=> DICT",
            list(value.keys())[:20]
        )
    else:
        print(
            key,
            "=>",
            type(value).__name__,
            repr(value)[:120]
        )
