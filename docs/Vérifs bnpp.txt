from pathlib import Path
import json

root = Path("portal/commun/data/standalone")

print("=" * 80)
print("1. FICHIERS JSON JIRA / PAYLOAD")
print("=" * 80)

for p in sorted(root.rglob("*.json")):
    s = str(p).lower()

    if "jira" not in s and "payload" not in s:
        continue

    try:
        d = json.loads(
            p.read_text(
                encoding="utf-8",
                errors="replace"
            )
        )
    except Exception:
        continue

    print()
    print("FICHIER =", p)

    if isinstance(d, dict):
        print("CLES =", list(d.keys())[:20])

        candidates = []

        for key, value in d.items():
            if isinstance(value, list):
                candidates.extend(value[:100])

    elif isinstance(d, list):
        print("LISTE DIRECTE")
        candidates = d[:100]

    else:
        continue

    found = []

    for r in candidates:
        if not isinstance(r, dict):
            continue

        txt = " ".join(
            str(v)
            for v in r.values()
            if isinstance(v, (str, int))
        )

        if (
            "[FEATURE GIL]" in txt
            or "Aliasing of internal IDs" in txt
            or "API2API" in txt
        ):
            found.append(r)

    if found:
        print(">>> FEATURES GIL TROUVEES =", len(found))

        for r in found[:10]:
