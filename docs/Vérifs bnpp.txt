python - <<'PY'
import json
from pathlib import Path

p = Path(
    "portal/commun/data/standalone/jira/"
    "capabilities.json"
)

d = json.loads(
    p.read_text(
        encoding="utf-8",
        errors="replace",
    )
)

rows = d.get("capabilities") or []

print("JIRA =", len(rows))

for r in rows:
    print()
    print("KEY        =", r.get("jiraKey"))
    print("CAPABILITY =", r.get("capability"))
    print("TITRE      =", r.get("titre"))
PY
