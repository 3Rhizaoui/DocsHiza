python - <<'PY'
import json
from pathlib import Path

p = Path(
    "portal/commun/data/standalone/jira/"
    "capabilities_brut.json"
)

data = json.loads(
    p.read_text(
        encoding="utf-8",
        errors="replace"
    )
)

for row in data.get("capabilities", []):

    epic = row.get("epic") or {}

    if epic.get("key") != "AERL_GIL-206":
        continue

    print("===================================")
    print("EPIC")
    print("===================================")
    print("KEY     =", epic.get("key"))

    ef = epic.get("fields") or {}

    for name in [
        "summary",
        "status",
        "resolution",
        "assignee",
        "reporter",
        "description",
        "fixVersions",
        "versions",
        "environment",
        "comment",
        "issuetype",
    ]:
        print()
        print(name, "=")
        print(
            json.dumps(
                ef.get(name),
                ensure_ascii=False,
                indent=2
            )
        )

    print()
    print("===================================")
    print("ENFANTS =", len(row.get("children") or []))
    print("===================================")

    for child in row.get("children") or []:

        fields = child.get("fields") or {}

        print()
        print("-----------------------------------")
        print("CHILD =", child.get("key"))
        print("-----------------------------------")

        for name in [
            "summary",
            "status",
            "resolution",
            "assignee",
            "reporter",
            "description",
            "fixVersions",
            "versions",
            "environment",
            "comment",
            "issuetype",
        ]:
            print(
                name,
                "=",
                json.dumps(
                    fields.get(name),
                    ensure_ascii=False
                )[:3000]
            )

    break
PY
