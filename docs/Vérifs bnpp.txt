import json
from pathlib import Path

p = Path(
    "portal/commun/data/standalone/jira/"
    "capabilities_brut.json"
)

data = json.loads(
    p.read_text(
        encoding="utf-8",
        errors="replace"
    )
)

for row in data.get("capabilities", []):

    epic = row.get("epic") or {}

    if epic.get("key") != "AERL_GIL-206":
        continue

    print("EPIC =", epic.get("key"))
    print(
        "SUMMARY =",
        (epic.get("fields") or {}).get("summary")
    )

    children = row.get("children") or []

    print("ENFANTS =", len(children))
    print()

    for child in children:

        fields = child.get("fields") or {}

        print(
            child.get("key"),
            "|",
            fields.get("summary"),
            "| statut =",
            (fields.get("status") or {}).get("name"),
            "| type =",
            (fields.get("issuetype") or {}).get("name")
        )

    print()
    print("DIAGNOSTICS =")

    for diag in (
        row.get("childrenDiagnostics")
        or []
    ):
        print(diag)

    break
PY
