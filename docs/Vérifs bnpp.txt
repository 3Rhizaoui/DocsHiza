Après authentification, vérifie seulement le brut réel :

python - <<'PY'
import json
from pathlib import Path

p = Path(
    "portal/commun/data/standalone/jira/"
    "capabilities_brut.json"
)

data = json.loads(
    p.read_text(
        encoding="utf-8",
        errors="replace"
    )
)

rows = data.get("capabilities") or []

print("CAPABILITIES =", len(rows))

for row in rows:
    epic = row.get("epic") or {}
    fields = epic.get("fields") or {}

    print(
        epic.get("key"),
        "|",
        fields.get("summary"),
        "| enfants =",
        len(row.get("children") or [])
    )
PY
